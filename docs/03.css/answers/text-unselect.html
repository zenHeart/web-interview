实现代码

/* CSS 禁用选择 */
.unselectable-text {
  -webkit-user-select: none;
  -moz-user-select: none;
  -ms-user-select: none;
  user-select: none;
}

/* JavaScript 阻止复制 */
document.addEventListener("copy", function (e) {
  e.preventDefault();
});
