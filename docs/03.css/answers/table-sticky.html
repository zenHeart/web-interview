JavaScript 代码：

window.onscroll = function() {
  var table = document.querySelector(".js-sticky-table");
  var rect = table.getBoundingClientRect();
  
  if (rect.top <= 0) {
    table.classList.add("sticky");
  } else {
    table.classList.remove("sticky");
  }
};
