  open({ title, content, onClose, onConfirm, onCancel } = {}) {
    let modal = document.createElement('dialog-modal');
    // 支持函数式内容（如 JSX 预渲染结果）
    let titleHtml = typeof title === 'function' ? title() : title || '';
    let contentHtml = typeof content === 'function' ? content() : content || '';

      <div class="modal-title">${titleHtml}</div>
      <div class="modal-content">${contentHtml}</div>
      <div class="modal-footer">
        <button id="confirm">确定</button>
        <button id="cancel">取消</button>
      </div>
    `;
    document.body.appendChild(modal);
    modal.open();
    // 关闭（遮罩/关闭按钮/ESC）
    const closeHandler = () => {
      modal.close();
      modal.remove();
      onClose && onClose();
    };
    modal.shadowRoot.getElementById('close').onclick = closeHandler;
    modal.shadowRoot.querySelector('.modal-mask').onclick = (e) => {
      if (e.target === e.currentTarget) closeHandler();
    };
    // 确定/取消
    modal.querySelector('#confirm').onclick = () => {
      onConfirm && onConfirm();
      modal.close();
      modal.remove();
    };
    modal.querySelector('#cancel').onclick = () => {
      onCancel && onCancel();
      modal.close();
      modal.remove();
    };
    return modal;
  }
};
</script>
