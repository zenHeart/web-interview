## Fizz Buzz 问题 {#p0-fizz-buzz}

```js
function fizzBuzz (n) {
  const result = []
  for (let i = 1; i <= n; i++) {
    if (i % 3 === 0 && i % 5 === 0) {
      result.push('FizzBuzz')
    } else if (i % 3 === 0) {
      result.push('Fizz')
    } else if (i % 5 === 0) {
      result.push('Buzz')
    } else {
      result.push(i.toString())
    }
  }
  return result
}
```

## 寻找首次匹配子序列 {#p0-find-subsequence}

```js
function findSubsequence (arr, subseq) {
  let i = 0
  for (let j = 0; j < arr.length; j++) {
    if (arr[j] === subseq[i]) {
      i++
    }
    if (i === subseq.length) {
      return j - i + 1
    }
  }
  return -1
}
```

## 移动零 {#p0-move-zero}

```js
function moveZeroes (nums) {
  let index = 0
  for (let i = 0; i < nums.length; i++) {
    if (nums[i] !== 0) {
      nums[index++] = nums[i]
    }
  }
  for (let i = index; i < nums.length; i++) {
    nums[i] = 0
  }
}
```
